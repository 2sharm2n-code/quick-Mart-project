const qs=id=>document.getElementById(id),esc=v=>String(v??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[c]));
const counties=['Nairobi','Mombasa','Nakuru','Uasin Gishu','Kiambu','Kisumu','Machakos','Kajiado','Nyeri','Meru'];
function fillSelect(id,items){qs(id).innerHTML='<option value="">Select</option>'+items.map(x=>`<option>${esc(x)}</option>`).join('')}
fillSelect('homeCounty',counties);fillSelect('branchCounty',counties);
qs('screeningQuestions').innerHTML=`<div class="screening-declaration"><b>Availability confirmation</b><p>HR will review the profile information above as part of the initial screening.</p><label class="checkline"><input id="availability" type="checkbox" required> I confirm that the information above is accurate and that I am available for the position requirements, including shifts where applicable.</label></div>`;
qs('phoneLocal').addEventListener('input',e=>e.target.value=e.target.value.replace(/\D/g,'').slice(0,9));
let positions=[];
async function loadPositions(){try{const r=await fetch('/api/positions',{cache:'no-store'}),d=await r.json();positions=(d.positions||[]).filter(p=>Number(p.slots_available)>0);qs('position').innerHTML='<option value="">Select an available position</option>'+positions.map(p=>`<option value="${esc(p.title)}">${esc(p.title)}</option>`).join('')||'<option value="">No positions available</option>';if(!positions.length)qs('positionInfo').innerHTML='<b>No public vacancies are open right now.</b> Please return later.'}catch{qs('position').innerHTML='<option value="">Could not load positions</option>'}}
qs('position').addEventListener('change',()=>{const p=positions.find(x=>x.title===qs('position').value);qs('positionInfo').innerHTML=p?`<b>${esc(p.title)}</b><br>${Number(p.slots_available).toLocaleString()} slot(s) currently available.<br>Salary will be shown for confirmation at the final Apply stage.`:'Select the position you want.'});
loadPositions();
qs('applyForm').onsubmit=async e=>{e.preventDefault();const b=e.currentTarget.querySelector('button');b.disabled=true;b.textContent='Submitting profile…';const phone='254'+qs('phoneLocal').value;try{const r=await fetch('/api/applications',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({fullName:qs('fullName').value.trim(),phone,email:qs('email').value.trim(),homeCounty:qs('homeCounty').value,preferredBranchCounty:qs('branchCounty').value,appliedPosition:qs('position').value,age:qs('age').value,educationLevel:qs('educationLevel').value,highestQualification:qs('highestQualification').value,relevantExperience:qs('relevantExperience').value,screeningAnswers:JSON.stringify({availability:qs('availability').checked?'Available for required shifts':'Not confirmed'})})});const d=await r.json();if(!r.ok){qs('applyResult').innerHTML=`<div class="error">${esc(d.message||'Screening could not be submitted.')}</div>`;return}location.href='/portal/'+encodeURIComponent(d.application.application_ref)}catch{qs('applyResult').innerHTML='<div class="error">Could not reach the recruitment server. Please try again.</div>'}finally{b.disabled=false;b.textContent='Finish screening →'}};
